import { Component, OnInit } from '@angular/core'; import { HttpServiceService } from '../http-service.service'; import { Router } from '@angular/router'; import { DataValidator } from '../utility/data-validator'; @Component({ selector: 'app-signup', templateUrl: './signup.component.html', styleUrls: ['./signup.component.css'] }) export class SignupComponent implements OnInit { form: any = { error: false, message: '', // ✅ HTML uses: form.inputerror.firstName, form.inputerror.password, etc. inputerror: {}, // ✅ HTML uses: form.data.firstName, form.data.login, form.data.password, etc. data: { id: null, firstName: '', lastName: '', login: '', password: '', phone: '', dob: '', gender: '' } }; constructor( private httpService: HttpServiceService, private dataValidator: DataValidator, private router: Router ) { } ngOnInit(): void { } validate() { return ( this.dataValidator.isNotNull(this.form.data.firstName) && this.dataValidator.isNotNull(this.form.data.lastName) && this.dataValidator.isNotNull(this.form.data.login) && this.dataValidator.isNotNull(this.form.data.password) && this.dataValidator.isNotNull(this.form.data.phone) && this.dataValidator.isNotNull(this.form.data.dob) && this.dataValidator.isNotNull(this.form.data.gender) ); } submit() { const _self = this; // ✅ Reset form messages this.form.error = false; this.form.message = ''; this.form.inputerror = {}; this.httpService.post("/Auth/signup", this.form.data, function (res) { if (res.result?.message) { _self.form.message = res.result.message; } if (res.result?.inputerror) { _self.form.inputerror = res.result.inputerror; } _self.form.error = !res.success; if (res.success) { _self.form.message = "Account Created Successfully"; _self.router.navigateByUrl('/login'); } }); } exit() { this.form = { error: false, message: '', inputerror: {}, data: { id: null, firstName: '', lastName: '', login: '', password: '', phone: '', dob: '', gender: '' } }; } }

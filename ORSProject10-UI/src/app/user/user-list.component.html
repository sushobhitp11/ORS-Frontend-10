import { Component, OnInit } from '@angular/core'; import { Router } from '@angular/router'; import { HttpServiceService } from '../http-service.service'; @Component({ selector: 'app-user-list', templateUrl: './user-list.component.html', styleUrls: ['./user-list.component.css'] }) export class UserlistComponent implements OnInit { // ✅ MASTER CHECKBOX isMasterSel: boolean = false; // ✅ NEXT PAGE BUTTON ENABLE/DISABLE nextList: number = 0; // ✅ COMPLETE FORM STRUCTURE (Jenkins Required) form: any = { error: false, message: '', pageNo: 0, searchParams: { firstName: '', roleId: null }, preload: { roleList: [] }, list: [] // ✅ User table data }; constructor(private router: Router, private httpService: HttpServiceService) {} ngOnInit() { this.preload(); this.search(); } // ✅ PRELOAD ROLE LIST preload() { this.httpService.get('/User/preload', res => { if (res.success) { this.form.preload.roleList = res.result.roleList; } }); } // ✅ SEARCH USERS search() { this.httpService.post('/User/search/' + this.form.pageNo, this.form.searchParams, res => { this.form.list = res.result.data; this.form.error = res.error; this.form.message = res.message; this.nextList = res.result.nextList; }); } // ✅ SEARCH BUTTON submit() { this.form.pageNo = 0; this.search(); } // ✅ PAGINATION previous() { if (this.form.pageNo > 0) { this.form.pageNo--; this.search(); } } next() { if (this.nextList !== 0) { this.form.pageNo++; this.search(); } } // ✅ OPEN USER FORM forward(url: string) { this.router.navigateByUrl(url); } // ✅ SELECT ALL CHECKBOX checkUncheckAll(event: any) { this.isMasterSel = event.target.checked; this.form.list.forEach((u: any) => { if (u.roleName !== 'Admin') { u.selected = this.isMasterSel; } }); } // ✅ UPDATE SINGLE CHECKBOX checklistUpdate() { let total = this.form.list.filter((u: any) => u.roleName !== 'Admin').length; let selected = this.form.list.filter((u: any) => u.selected).length; this.isMasterSel = (total === selected); } // ✅ DELETE SELECTED deleteMany() { let ids: any[] = []; this.form.list.forEach((u: any) => { if (u.selected === true) { ids.push(u.id); } }); if (ids.length === 0) { this.form.error = true; this.form.message = "Please select at least one user!"; return; } this.httpService.post('/User/deleteMany', { ids: ids }, res => { this.form.error = res.error; this.form.message = res.message; this.search(); }); } // ✅ RESET SEARCH exit() { this.form.searchParams = { firstName: '', roleId: null }; this.submit(); } }
